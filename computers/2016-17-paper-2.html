QUESTION 1
Write a program to accept two dates and the weekday name of the 1st day of the months to find out the number of days between them (Include start date and Exclude end date in calculation). Also generate the calendar of the given months. Valid dates should be checked for.

Answer

Answer as the practical expects it: the algorithm, then the documented Java program, then the output for the paper's example and for further inputs — every output below is from an actual run of this program (JDK 11).

Idea. Turn each date into a serial day number (days from 1/1/0001 up to and including it, counting leap years); the difference end − start counts the start date and excludes the end date: 10/8 → 16/11/2016 is 22 (Aug 10–31) + 30 + 31 + 15 (Nov 1–15) = 98. A date is valid if 1 ≤ month ≤ 12 and 1 ≤ day ≤ days in that month (29 in February of a leap year). The calendars are printed from the weekday given for the 1st of each month. If both dates fall in the same month its calendar is printed once.

Algorithm

  1. Start. Input start date and end date as dd/mm/yyyy; split each at '/'.
  2. If either is not a valid date (non-numeric, month outside 1–12, day outside 1–days(month, year)) print "INVALID DATE" and stop.
  3. serial(d, m, y) ← Σ days of years 1 … y−1 + Σ days of months 1 … m−1 of year y + d.
  4. diff ← serial(end) − serial(start). If diff ≤ 0 print "INVALID INPUT" and stop.
  5. Input the weekday name of the 1st of the start month (and of the end month if it is a different month); convert to 0 (Sunday) … 6 (Saturday); if not a weekday name print "INVALID WEEKDAY" and stop.
  6. Print "No of days between start date and end date: diff days".
  7. For each month: print the month and year and the heading SUN … SAT; print f blank cells (f = weekday of the 1st); print 1 … days(month) in cells, starting a new line (and a dashed rule) after every 7th cell.
  8. Stop.

Program

import java.util.Scanner;

/* Reads two dates (dd/mm/yyyy) and the weekday of the 1st of each date's month,
 * prints the number of days from the start date (counted) to the end date
 * (not counted) and the calendar of both months.                            */
class DateCalendar {
    int mdays[] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    String mname[] = {"", "January", "February", "March", "April", "May", "June", "July",
                      "August", "September", "October", "November", "December"};
    String wk[] = {"SUNDAY", "MONDAY", "TUESDAY", "WEDNESDAY", "THURSDAY", "FRIDAY", "SATURDAY"};

    boolean isLeap(int y) {
        return (y % 400 == 0) || (y % 4 == 0 && y % 100 != 0);
    }

    int daysIn(int m, int y) {
        return (m == 2 && isLeap(y)) ? 29 : mdays[m];
    }

    // splits "dd/mm/yyyy" into {d, m, y}; returns null if it is not a valid date
    int[] parse(String s) {
        String p[] = s.trim().split("/");
        if (p.length != 3)
            return null;
        int d, m, y;
        try {
            d = Integer.parseInt(p[0].trim());
            m = Integer.parseInt(p[1].trim());
            y = Integer.parseInt(p[2].trim());
        } catch (NumberFormatException e) {
            return null;
        }
        if (y < 1 || m < 1 || m > 12 || d < 1 || d > daysIn(m, y))
            return null;
        return new int[]{d, m, y};
    }

    // number of days from 1/1/0001 up to and including the given date
    long serial(int d, int m, int y) {
        long n = 0;
        for (int yr = 1; yr < y; yr++)
            n += isLeap(yr) ? 366 : 365;
        for (int mo = 1; mo < m; mo++)
            n += daysIn(mo, y);
        return n + d;
    }

    // weekday name -> 0 (Sunday) .. 6 (Saturday); -1 if not a weekday
    int dayNo(String s) {
        for (int i = 0; i < 7; i++)
            if (wk[i].equalsIgnoreCase(s.trim()))
                return i;
        return -1;
    }

    void line() {
        System.out.println("-----------------------------------");
    }

    // calendar of month m of year y whose 1st falls on weekday f
    void calendar(int m, int y, int f) {
        line();
        System.out.println("           " + mname[m] + " " + y);
        line();
        System.out.println("SUN  MON  TUE  WED  THU  FRI  SAT");
        line();
        int col = 0;
        for (int i = 0; i < f; i++) {          // blank cells before the 1st
            System.out.print("     ");
            col++;
        }
        for (int d = 1; d <= daysIn(m, y); d++) {
            System.out.print(String.format("%-5d", d));
            if (++col == 7) {                  // end of a week
                System.out.println();
                line();
                col = 0;
            }
        }
        if (col != 0) {
            System.out.println();
            line();
        }
        System.out.println();
    }

    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        DateCalendar ob = new DateCalendar();
        System.out.println("Start Date (dd/mm/yyyy):");
        int s[] = ob.parse(sc.nextLine());
        System.out.println("End Date (dd/mm/yyyy):");
        int e[] = ob.parse(sc.nextLine());
        if (s == null || e == null) {
            System.out.println("INVALID DATE");
            return;
        }
        long diff = ob.serial(e[0], e[1], e[2]) - ob.serial(s[0], s[1], s[2]);
        if (diff <= 0) {
            System.out.println("INVALID INPUT: end date must be after the start date");
            return;
        }
        boolean same = (s[1] == e[1] && s[2] == e[2]);     // both dates in one month
        System.out.println("1st day of " + ob.mname[s[1]] + ":");
        int f1 = ob.dayNo(sc.nextLine());
        int f2 = f1;
        if (!same) {
            System.out.println("1st day of " + ob.mname[e[1]] + ":");
            f2 = ob.dayNo(sc.nextLine());
        }
        if (f1 < 0 || f2 < 0) {
            System.out.println("INVALID WEEKDAY");
            return;
        }
        System.out.println("No of days between start date and end date: " + diff + " days");
        ob.calendar(s[1], s[2], f1);
        if (!same)
            ob.calendar(e[1], e[2], f2);
    }
}

Q2Methods (number programs)[20]

QUESTION 2
Write a Program in Java to input a number and check whether it is a Bouncy Number or not.

Increasing Number: Working from left-to-right if no digit is exceeded by the digit to its left it is called an increasing number; for example, 556778.

Decreasing Number: Similarly if no digit is exceeded by the digit to its right it is called a decreasing number; for example, 663320.

Bouncy Number: A positive integer that is neither increasing nor decreasing is a "bouncy" number; for example, 29748. There cannot be any bouncy number below 100.

Answer

Answer as the practical expects it: the algorithm, then the documented Java program, then the output for the paper's example and for further inputs — every output below is from an actual run of this program (JDK 11).

Algorithm

  1. Start. Input n; if n ≤ 0 print "INVALID INPUT" and stop. s ← n as a string.
  2. inc ← true; for i = 1 to len(s) − 1: if s[i] < s[i−1], inc ← false.
  3. dec ← true; for i = 0 to len(s) − 2: if s[i] < s[i+1], dec ← false.
  4. If not inc and not dec, print "n is a Bouncy Number"; otherwise print "n is NOT a Bouncy Number" (with the reason: increasing, decreasing, or all digits equal). Stop.

Program

import java.util.Scanner;

/* Increasing: no digit is smaller than the digit on its left  (556778)
 * Decreasing: no digit is smaller than the digit on its right (663320)
 * Bouncy    : a positive integer that is neither                (29748)  */
class Bouncy {
    long num;       // the number to test

    Bouncy(long n) {
        num = n;
    }

    boolean isIncreasing() {
        String s = Long.toString(num);
        for (int i = 1; i < s.length(); i++)
            if (s.charAt(i) < s.charAt(i - 1))    // a digit exceeded by the one on its left
                return false;
        return true;
    }

    boolean isDecreasing() {
        String s = Long.toString(num);
        for (int i = 0; i < s.length() - 1; i++)
            if (s.charAt(i) < s.charAt(i + 1))    // a digit exceeded by the one on its right
                return false;
        return true;
    }

    boolean isBouncy() {
        return !isIncreasing() && !isDecreasing();
    }

    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter a positive integer:");
        long n = sc.nextLong();
        if (n <= 0) {
            System.out.println("INVALID INPUT: enter a positive integer");
            return;
        }
        Bouncy ob = new Bouncy(n);
        if (ob.isBouncy())
            System.out.println(n + " is a Bouncy Number");
        else if (ob.isIncreasing() && ob.isDecreasing())
            System.out.println(n + " is NOT a Bouncy Number (all its digits are equal)");
        else if (ob.isIncreasing())
            System.out.println(n + " is NOT a Bouncy Number (it is an increasing number)");
        else
            System.out.println(n + " is NOT a Bouncy Number (it is a decreasing number)");
    }
}

Q3Arrays (single-dimensional)[20]

QUESTION 3
Given an array of integers, find the indices of the pair of integers that have the smallest difference between them based on the following conditions:

  1. The actual difference between two elements (not their indices)
  2. The pair with the lower first index (if the difference is the same between two pairs).
  3. The pair with the lower second index (if the first indices are the same between two pairs and if the difference between the elements is same).

Sample Input 1
Consider an array {5, 3, 21, 10, 12, 7}
The distance between various pair of elements is shown in the following figure.

The pairs [5,3], [10,12] have the difference 2. But the minimum valued pair is [5,3]. Therefore the index of element 5 is 0 and 3 is 1.

Sample Output 1
The indices are 0, 1.

Sample Input 2
Consider an array {19, 0, -5, 4, 7, 10}
The distance between the pair of elements is shown in the following figure.

The pairs [4,7], [7,10] have the difference 3. But the minimum valued pair is [4,7]. Therefore the index of element 4 is 3 and 7 is 4.

Sample Output 2
The indices are 3, 4.

Answer

Answer as the practical expects it: the algorithm, then the documented Java program, then the output for the paper's example and for further inputs — every output below is from an actual run of this program (JDK 11).

Reading the problem. The tables show only neighbouring pairs, but condition 3 ("the lower second index if the first indices are the same") only makes sense if any two elements may form a pair, so all pairs (i, j) with i < j are compared, using the absolute difference. Sample 1 itself needs rule 3: over all pairs, 5 and 7 (indices 0, 5) also differ by 2, and rule 3 picks 0, 1 over 0, 5. Both samples give the printed answer. Scanning i upwards and, for each i, j upwards, and replacing the best pair only on a strictly smaller difference, automatically applies the tie-break rules 2 and 3.

Algorithm

  1. Start. Input n; if n < 2 print "INVALID INPUT" and stop. Input a[0..n−1].
  2. best ← |a[0] − a[1]|, bi ← 0, bj ← 1.
  3. For i = 0 to n−2, for j = i+1 to n−1: d ← |a[i] − a[j]|; if d < best then best ← d, bi ← i, bj ← j.
  4. Print "The indices are bi, bj." Stop.

Program

import java.util.Scanner;

/* Finds the pair (i, j), i < j, whose elements differ the least.
 * Pairs are examined with i increasing and, for each i, j increasing, and the
 * stored pair is replaced only by a STRICTLY smaller difference.  So on a tie
 * the pair with the lower first index is kept, and if the first indices are
 * equal, the one with the lower second index -- exactly conditions 2 and 3.  */
class MinDiff {
    int a[];        // the array
    int n;          // its size

    MinDiff(int size) {
        n = size;
        a = new int[n];
    }

    void accept(Scanner sc) {
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();
    }

    // returns {i, j} of the pair with the smallest difference
    int[] closest() {
        int bi = 0, bj = 1;
        int best = Math.abs(a[0] - a[1]);
        for (int i = 0; i < n - 1; i++)
            for (int j = i + 1; j < n; j++) {
                int d = Math.abs(a[i] - a[j]);     // actual difference of the elements
                if (d < best) {                    // strictly smaller only
                    best = d;
                    bi = i;
                    bj = j;
                }
            }
        return new int[]{bi, bj};
    }

    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        System.out.println("Number of elements:");
        int size = sc.nextInt();
        if (size < 2) {
            System.out.println("INVALID INPUT: at least two elements are needed");
            return;
        }
        MinDiff ob = new MinDiff(size);
        System.out.println("Enter the elements:");
        ob.accept(sc);
        int p[] = ob.closest();
        System.out.println("The indices are " + p[0] + ", " + p[1] + ".");
    }
}
